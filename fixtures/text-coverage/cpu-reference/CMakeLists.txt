cmake_minimum_required(VERSION 3.20)
project(dinf_cpu_flash_probe LANGUAGES C CXX)
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0)
find_package(ggml REQUIRED CONFIG PATHS "${DINF_GGML_PREFIX}/lib/cmake/ggml" NO_DEFAULT_PATH)
if(NOT GGML_VERSION STREQUAL "0.10.0")
  message(FATAL_ERROR "Only ggml0.10.0 is supported")
endif()
add_library(ggml-base ALIAS ggml::ggml-base)
set(BUILD_SHARED_LIBS ON)
set(GGML_SYSTEM_ARCH ARM)
set(GGML_NATIVE OFF)
set(GGML_CPU_ALL_VARIANTS ON)
set(GGML_CPU_ARM_ARCH "")
set(GGML_INTERNAL_DOTPROD ON)
set(GGML_INTERNAL_MATMUL_INT8 ON)
set(GGML_INTERNAL_NOSVE ON)
set(GGML_INTERNAL_SME ON)
set(CMAKE_BUILD_RPATH "${DINF_GGML_PREFIX}/lib")
include(CheckCXXCompilerFlag)
include(CheckIPOSupported)
check_ipo_supported(RESULT supports_lto OUTPUT lto_error)
if(NOT supports_lto)
  message(FATAL_ERROR "Installed ggml enables LTO; local compiler must support it: ${lto_error}")
endif()
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)
add_compile_definitions(_XOPEN_SOURCE=600 _DARWIN_C_SOURCE GGML_SCHED_MAX_COPIES=4)
function(ggml_add_backend_library backend)
  add_library(${backend} MODULE)
  set_target_properties(${backend} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
  target_compile_definitions(${backend} PRIVATE GGML_BACKEND_DL GGML_BACKEND_BUILD GGML_BACKEND_SHARED)
  target_link_libraries(${backend} PRIVATE ggml-base)
  target_include_directories(${backend} PRIVATE .. ../include)
endfunction()
include(ggml-cpu/CMakeLists.txt)
ggml_add_cpu_backend_variant_impl(apple_m4)
if(NOT GGML_OPENMP_ENABLED)
  message(FATAL_ERROR "Installed ggml enables OpenMP; no silent local fallback allowed")
endif()
